\chapter{Introduction}
Language is, perhaps, one of the most fundamental aspects of human society. Beyond serving practically as a means of communication, languages embody culture, history and collective identity, preserving traditions, belief systems and social structures that have developed over generations. The modern world contains thousands of languages with a remarkable range of similarities and differences. By investigating how languages change and diversify over time, historical linguistics provides a window into the development of human societies themselves.

The task of reconstructing and explaining the evolution of all languages is no trivial undertaking. Szathmáry and Smith (1995) posit that language emergence in human societies is one of the major transitions in evolution \cite{Szathmary1995}, while Christiansen and Kirby (2003) describe languages as "by far the most complex behaviour we know of" \cite{Christiansen2003}. Languages undergo evolution at varying timescales \cite{Christiansen2003} while importing words from other languages (or \textit{borrowing}) is commonplace \cite{Campbell2004}. On top of that, there are currently over 7000 languages in the world \cite{Ethnologue}. Furthermore, direct evidence about languages is not readily available and hypotheses about language evolution must be tested through an interdisciplinary analysis \cite{Christiansen2003}.

Despite these challenges, there have been many approaches to reconstruct language evolution. A common starting point is the observation that many languages exhibit systematic similarities, suggesting a shared common historical origin between them. As such, identifying and classifying these relationships has become a central objective of historical linguistics. Parsons (1767) first considered such similarities of numerals of languages in Europe and India, concluding that the languages must be related via a common ancestor \cite{Parsons1767}. These languages would later be proven to be from the same \textit{language family} (the \textit{Indo-European} family) \cite{Bopp1816} - where groups of languages which are said to derive from a common language are \textit{genetically} related via a language family \cite{Rowe2015}. Today, there are now as many as 143 recognised language families \cite{Ethnologue}, similarly identified via studying a plethora of similarities and systematic changes in languages over time \cite{Campbell2004} as well as employing more computational and quantitative methods in the age of big data \cite{Bhattacharya2018}. These methods, which are commonly imported from other disciplines like phylogenetics, need to be carefully considered in the context of historical linguistics.

This chapter introduces the principal statistical methods used in historical linguistics, culminating in the advent of phylogenetic analysis in language evolution. Particular attention is given to the assumptions of the model, especially the assumption of site independence, and the extent of which these assumptions might be violated in linguistic data. Relevant previous work concerning site independence in both phylogenetics and linguistics is then reviewed.

More sophisticated methods have since been used to justify systematic similarities and differences between languages for a given meaning \cite{Campbell2004}. While the exact methods used by linguists is outside the scope of this paper, we note that for a given meaning and set of languages, we can group languages together based on how similar the words (or \textit{cognates}) used are. We define several terms to formalise our analysis:

\begin{definition} [Cognate]
    Consider words (or more generally, lexical items) $w_1^m \in L_1$ and $w_2^m \in L_2$, from languages $L_1$ and $L_2$ respectively, that have meaning $m$. We can define a binary relation $\sim$ representing similarity of words of a given meaning $m$, such that if $w_1^m \sim w_2^m$, we say that $w_1^m$ and $w_2^m$ are cognates.
\end{definition}

\begin{definition} [Cognacy class]
\label{def:cognacyclass}
    Given meaning $m$ and words $w_1^m \in L_1, \dots, w_n^m \in L_n$, the cognacy classes of a given meaning $m$ are the sets $\{P_1^m, \dots, P_K^m\} =: P$ of $\{w_1^m, \dots w_n^m\}$ that represent the equivalence classes of $\sim$ for a given meaning $m$. In other words, if $P$ satisfies:
    \begin{enumerate}
        \item $\forall k = 1, \dots, K, \forall w_i^m, w_j^m \in P_k^m, w_i^m \sim w_j^m$ and
        \item $\forall k, l = 1, \dots, K, l \neq j, \forall w_i^m \in P_k^m, w_j^m \in P_l^m, w_i^m \nsim w_j^m$,
    \end{enumerate}
    then $P_1^m, \dots, P_K^m$ are cognacy classes.
\end{definition}

We note that under the assumption where all languages have exactly one word to describe meaning $m$, $P$ is exactly a partition. However, there are numerous cases where a language has more than one word to describe meaning $m$ which may correspond to multiple cognacy classes, and so we accommodate for this in the definition above. Clearly, the value of $K$ in Definition \ref{def:cognacyclass} can vary between $1$ (all languages have similar words for meaning $m$) and $n$ (no languages have similar words for meaning $m$). Each cognacy class can then be used to encode a vector of languages that belong to the cognacy class, which we will define as a site:

\begin{definition} [Site]
    Given list of languages $\{L_1, ..., L_n\}$ and cognacy class $P_k^m := \{w_{i_1}^m, \dots, w_{i_P}^m \}$, where $w_{i_1}^m \in L_{i_1}, \dots w_{i_P}^m \in L_{i_P}$ and $1 \leq i_p \leq n, p = 1, \dots, P$. The corresponding site for cognacy class $P_k^m$ is the vector $\mathbf{s} \in [0, 1]^n$ such that $\forall j = 1, ..., n$:
    \begin{equation}
    \notag
        s_j = \begin{cases}
            1 & j \in \{i_1, ..., i_P\} \\
            0 & \text{otherwise}
        \end{cases}.
    \end{equation}
\end{definition}

As such, for a given meaning $m$, there are $K$ sites corresponding to the $K$ cognacy classes present amongst the $n$ languages. We can compile all sites across all meanings $m$ to form a binary data matrix $\mathbf{D} \in [0, 1]^{n \times S}$, where $S$ is the total number of sites, encoding all similarities and differences between languages $L_1, \dots, L_n$. Under such a construction, it is clear that the similarity of rows $i$ and $j$ of $\mathbf{D}$ (which we will call sequences) corresponds to how similar the languages are, given a set of meanings. \subsection{Language Trees}
Such relationships between languages are represented, most commonly, in the form of a language tree \cite{Pellard2026}, popularised by Schleicher (1861-62) \cite{Schleicher1861}. 

\begin{definition} [Language Tree]
    Let $T = (V, E, r)$ be a rooted directed tree, where $V$ is a finite set of vertices, $E \subseteq V \times V$ is a set of directed edges and $r \in V$ is a designated root node. In the context of historical linguistics, $T$ is a language tree if the underlying undirected tree is connected and acyclic (it is a directed acyclic graph), and every vertex $v \in V, v \neq r$ has exactly one parent $u \in V$.
\end{definition}

\begin{definition} [Observed and Unobserved Languages]
    Let $T = (V, E, r)$ be a language tree. The set of vertices $L := \{v \in V \ | \ \deg^+(v) = 0\}$ with outdegree zero (also known as the leaves or terminal nodes) as the observed languages of the tree. Additionally, the remaining vertices $W := V \setminus L$ represent the unobserved proto-languages that gave rise to events of divergence between languages.
\end{definition}

\begin{definition} [Clade]
    Let $T = (V, E, r)$ be a language tree. For some node $v \in V$, define its descendants as $\text{Desc}(v) := \{u \in V \ | \ v \to u \}$. A clade $C(v)$ with ancestor $v$ is defined as $C(v) = \text{Desc}(v) \ \cup \ {v}$, a subgroup including ancestor $v$ and all its descendants. 
\end{definition}

While other models exist, the tree model has persisted (with good reason) as the primary method in historical linguistics for representing language evolution. {\color{red} paraphrase this, this is directly lifted:} The model assumes linguistic features are inherited with modifications over time, while related languages descend from a single common ancestor via divergence \cite{Pellard2026}. The tree model can be further constrained such that the tree has a \textit{bifurcating} structure and has branches which have lengths proportional to time. This makes comparing events of divergence relevant: the tree model is able to hypothesise when two languages first diverged, and quantify how closely related languages are to each other. Although it is, like any model, unable to represent the full complexity of linguistic change, the model is well suited to cases of well-differentiated and geographically isolated languages \cite{Pellard2026}, thus being applicable to studying language dissimilarity within a language family.

\KwInput{$\textbf{m}$, such that $\mathbf{m}_i$ is the position of the $i$'th monitor\newline
$\textbf{l}$, such that $\mathbf{l}_i$ is the position of the $i$'th landmark\newline
$\mathbf{p}^m$, such that $\mathbf{p}^m_i$ is the ping latency from monitor $i$ to the target\newline
$\mathbf{p}^l$, such that $\mathbf{p}^l_i$ is the set of ping latencies to landmark $i$}

\BlankLine
\KwPre{Compute $\hat{p}_m(d \mid l)$, an estimator giving the likelihood of the target being distance $d$ away from the monitor, given that the monitor records a latency of $l$ to that target. Implemented by training a KDE using $\mathbf{p}^l$.\newline
Compute $\hat{p}_l(d \mid v)$, an estimator giving the likelihood of the target being distance $d$ away from the landmark, given a Canberra distance of $v$ between the target and the landmark, using training targets.
}
\BlankLine
\KwOutput{Most likely location of the target}
\BlankLine

\Fn{Likelihood($x$, $\mathbf{v}$)} {
MonitorScore $\gets \sum_{i=0}^{|\mathbf{m}|} \log{\hat{p}_m(d(x, \mathbf{m}_i) \mid l_i)}$\;
LandmarkScore $\gets \sum_{i=0}^{|\mathbf{l}|} \log{\hat{p}_l(d(x, \mathbf{l}_i) \mid \mathbf{v}_i)}$\;
\Return MonitorScore + LandmarkScore
}

\BlankLine
$\mathbf{v} \gets $\{$\mathrm{canberra\_distance}(\mathbf{l}_i, \mathbf{p}^m) \mid \mathbf{l}_i \in \mathbf{l}$\}

$\mathbf{C}$ $\gets$ Constraint-Based-Geolocation($\mathbf{m}$, $\mathbf{p}^m$)\;
$\mathbf{C_l}$ $\gets$ \{$m \in \mathbf{m} \mid \mathbf{C}$ contains $m\} \cup \{l \in \mathbf{l} \mid \mathbf{C}$ contains $l$\}\;
\BlankLine
\Return argmax$_{x\in \mathbf{C_l}}$ Likelihood($x$)
